Lớp chi tiết RTL
wire [31:0] regn_d, rd0, rd1, rd2, rd3, rd_mux;

[blk_wr_dec]
wr     = psel & penable & pwrite;
wr0    = wr & (paddr == 4'h0);
wr1    = wr & (paddr == 4'h4);
wr2    = wr & (paddr == 4'h8);
wr3    = wr & (paddr == 4'hC);
regn_d = pwdata;

[reg_data]
reg0_q <= regn_d @(rst=32'h00000000, en=wr0);
reg1_q <= regn_d @(rst=32'h00000000, en=wr1);
reg2_q <= regn_d @(rst=32'h00000000, en=wr2);
reg3_q <= regn_d @(rst=32'h00000000, en=wr3);

[reg_written]
reg0_written_q <= 1'b1 @(rst=1'b0, en=wr0);
reg1_written_q <= 1'b1 @(rst=1'b0, en=wr1);
reg2_written_q <= 1'b1 @(rst=1'b0, en=wr2);
reg3_written_q <= 1'b1 @(rst=1'b0, en=wr3);

[blk_rd_mux]
rd     = psel & penable & (~pwrite);
rd0    = reg0_written_q ? reg0_q : 32'hDEADBEEF;
rd1    = reg1_written_q ? reg1_q : 32'hDEADBEEF;
rd2    = reg2_written_q ? reg2_q : 32'hDEADBEEF;
rd3    = reg3_written_q ? reg3_q : 32'hDEADBEEF;
rd_mux = case (paddr)
           4'h0: rd0;
           4'h4: rd1;
           4'h8: rd2;
           4'hC: rd3;
           default: 32'h00000000;
         endcase;
prdata = rd ? rd_mux : 32'h00000000;

[blk_pready]
pready = 1'b1;
